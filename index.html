class CSMPointJacobian:
    def __init__(self, X: int, Y: int, Z: int):
        self.X, self.Y, self.Z = X % P, Y % P, Z % P

    def to_affine(self):
        if self.Z == 0: return None
        z_inv = pow(self.Z, P - 2, P)
        return ((self.X * z_inv**2) % P, (self.Y * z_inv**3) % P)

def csm_add(p1, p2):
    # Complete 12-Step Unrolled Jacobian Addition
    Z1_2, Z2_2 = (p1.Z**2) % P, (p2.Z**2) % P
    U1, U2 = (p1.X * Z2_2) % P, (p2.X * Z1_2) % P
    S1, S2 = (p1.Y * p2.Z * Z2_2) % P, (p2.Y * p1.Z * Z1_2) % P
    H = (U2 - U1) % P
    I = (2*H)**2 % P
    J = (H * I) % P
    r = 2 * (S2 - S1) % P
    V = (U1 * I) % P
    X3 = (r**2 - J - 2*V) % P
    Y3 = (r*(V - X3) - 2*S1*J) % P
    Z3 = (((p1.Z + p2.Z)**2 - Z1_2 - Z2_2) * H) % P
    return CSMPointJacobian(X3, Y3, Z3)
